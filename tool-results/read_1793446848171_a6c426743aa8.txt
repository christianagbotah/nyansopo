   326→function CollectDialog({ student, date, online, onOpenChange, onDone }: { student: PosStudent | null; date: string; online: boolean; onOpenChange: (o: boolean) => void; onDone: (receipt: any) => void }) {
   327→  const { currentSchoolId } = useStore();
   328→  const [amount, setAmount] = useState("");
   329→  const [method, setMethod] = useState("MOMO_MTN");
   330→  const [reference, setReference] = useState("");
   331→  const [busy, setBusy] = useState(false);
   332→
   333→  useEffect(() => { if (student) { setAmount((student.balance / 100).toFixed(2)); setReference(""); } }, [student]);
   334→
   335→  async function submit() {
   336→    if (!student || !currentSchoolId) return;
   337→    const cents = Math.round(parseFloat(amount || "0") * 100);
   338→    if (!cents || cents <= 0) { toast({ title: "Enter a valid amount" }); return; }
   339→    setBusy(true);
   340→    try {
   341→      const payload = { action: "pos-collect", studentId: student.id, amountCents: cents, method, reference: reference || undefined, date };
   342→      const res = await apiPostQueueable<any>(
   343→        `/api/school/${currentSchoolId}/ops/daily`, "payment", payload,
   344→        `Daily services ${fmtCedi(cents)} — ${student.name}`,
   345→        (p, clientId) => ({ ...p, category: "DAILY", schoolId: currentSchoolId, studentId: student.id, amountCents: cents, method, paidAt: new Date().toISOString(), clientId })
   346→      );
   347→      if (res.queued) {
   348→        toast({ title: "Queued offline", description: "Receipt will be issued when you reconnect." });
   349→        onDone(null);
   350→      } else {
   351→        onDone({ ...res.data?.payment, student: student.name });
   352→      }
   353→    } catch (e: any) { toast({ title: "Collection failed", description: e.message, variant: "destructive" }); } finally { setBusy(false); }
   354→  }
   355→
   356→  return (
   357→    <Dialog open={!!student} onOpenChange={onOpenChange}>
   358→      <DialogContent>
   359→        <DialogHeader>
   360→          <DialogTitle className="flex items-center gap-2"><Wallet className="size-4" /> Collect daily payment — {student?.name}</DialogTitle>
   361→          <DialogDescription>
   362→            {student?.classroom} · outstanding <b className="text-destructive">{student ? fmtCedi(student.balance) : ""}</b>
   363→            {!online && " · offline — will queue & sync"}
   364→          </DialogDescription>
   365→        </DialogHeader>
   366→        <div className="grid grid-cols-2 gap-3">
   367→          <div className="space-y-1.5">
   368→            <Label>Amount (GH₵) *</Label>
   369→            <Input type="number" step="0.01" value={amount} onChange={(e) => setAmount(e.target.value)} />
   370→          </div>
   371→          <div className="space-y-1.5">
   372→            <Label>Method</Label>
   373→            <Select value={method} onValueChange={setMethod}>
   374→              <SelectTrigger><SelectValue /></SelectTrigger>
   375→              <SelectContent>
   376→                {Object.entries(PAYMENT_LABELS).map(([k, v]) => <SelectItem key={k} value={k}>{v}</SelectItem>)}
   377→              </SelectContent>
   378→            </Select>
   379→          </div>
   380→          <div className="space-y-1.5 col-span-2">
   381→            <Label>MoMo / reference</Label>
   382→            <Input placeholder="e.g. MP2401.5533.77" value={reference} onChange={(e) => setReference(e.target.value)} />
   383→          </div>

   417→function ServiceDialog({ service, onOpenChange, onDone }: { service: Service | "new" | null; onOpenChange: (o: boolean) => void; onDone: () => void }) {
   418→  const { currentSchoolId } = useStore();
   419→  const [name, setName] = useState("");
   420→  const [category, setCategory] = useState("MEAL");
   421→  const [price, setPrice] = useState("");
   422→  const [chargeMode, setChargeMode] = useState("DAILY");
   423→  const [weekdays, setWeekdays] = useState<string[]>(["1", "2", "3", "4", "5"]);
   424→  const [busy, setBusy] = useState(false);
   425→
   426→  useEffect(() => {
   427→    if (service && service !== "new") {
   428→      setName(service.name); setCategory(service.category); setPrice((service.price / 100).toFixed(2));
   429→      setChargeMode(service.chargeMode); setWeekdays(service.weekdays.split(","));
   430→    } else if (service === "new") {
   431→      setName(""); setCategory("MEAL"); setPrice(""); setChargeMode("DAILY"); setWeekdays(["1", "2", "3", "4", "5"]);
   432→    }
   433→  }, [service]);
   434→
   435→  async function submit() {
   436→    if (!currentSchoolId || !name || !price) { toast({ title: "Name and price are required" }); return; }
   437→    setBusy(true);
   438→    try {
   439→      await apiPost(`/api/school/${currentSchoolId}/ops/daily`, {
   440→        action: "save-service",
   441→        id: service && service !== "new" ? service.id : undefined,
   442→        name, category, priceCents: Math.round(parseFloat(price) * 100), chargeMode,
   443→        weekdays: (chargeMode === "WEEKLY" ? weekdays.slice(0, 1) : weekdays).join(","),
   444→      });
   445→      toast({ title: service === "new" ? "Service created" : "Service updated" });
   446→      onDone();
   447→    } catch (e: any) { toast({ title: "Failed", description: e.message, variant: "destructive" }); } finally { setBusy(false); }
   448→  }
   449→
   450→  const toggleDay = (d: string) => setWeekdays((w) => (w.includes(d) ? w.filter((x) => x !== d) : [...w, d]));
   451→
   452→  return (
   453→    <Dialog open={!!service} onOpenChange={onOpenChange}>
   454→      <DialogContent>
   455→        <DialogHeader>
   456→          <DialogTitle>{service === "new" ? "New daily service" : "Edit service"}</DialogTitle>
   457→          <DialogDescription>e.g. Lunch GH₵8 every school day, or Sachet Water GH₵1 on Wednesdays only.</DialogDescription>
   458→        </DialogHeader>
   459→        <div className="grid grid-cols-2 gap-3">
   460→          <div className="space-y-1.5 col-span-2">
   461→            <Label>Name *</Label>
   462→            <Input value={name} onChange={(e) => setName(e.target.value)} placeholder="e.g. Lunch (canteen)" />
   463→          </div>
   464→          <div className="space-y-1.5">
   465→            <Label>Category</Label>
   466→            <Select value={category} onValueChange={setCategory}>
   467→              <SelectTrigger><SelectValue /></SelectTrigger>
   468→              <SelectContent>
   469→                <SelectItem value="MEAL">Meal</SelectItem><SelectItem value="DRINK">Drink</SelectItem>
   470→                <SelectItem value="CLASS">Class / remedial</SelectItem><SelectItem value="CARE">Care</SelectItem>
   471→                <SelectItem value="OTHER">Other</SelectItem>
   472→              </SelectContent>
   473→            </Select>
   474→          </div>
   475→          <div className="space-y-1.5">
   476→            <Label>Price (GH₵) *</Label>
   477→            <Input type="number" step="0.5" value={price} onChange={(e) => setPrice(e.target.value)} placeholder="8.00" />
   478→          </div>
   479→          <div className="space-y-1.5 col-span-2">
   480→            <Label>Charging pattern</Label>
   481→            <Select value={chargeMode} onValueChange={setChargeMode}>
   482→              <SelectTrigger><SelectValue /></SelectTrigger>
   483→              <SelectContent>
   484→                <SelectItem value="DAILY">Daily — every selected school day</SelectItem>
   485→                <SelectItem value="WEEKLY">Weekly — once a week</SelectItem>
   486→              </SelectContent>
   487→            </Select>
   488→          </div>
   489→          <div className="space-y-1.5 col-span-2">
   490→            <Label>Days it runs {chargeMode === "WEEKLY" && <span className="text-muted-foreground font-normal">(pick one day)</span>}</Label>
   491→            <div className="flex gap-1.5 flex-wrap">
   492→              {DAY_LABELS.slice(1, 6).map((d, i) => (
   493→                <button key={d} type="button"
   494→                  onClick={() => toggleDay(String(i + 1))}
   495→                  className={`size-9 rounded-lg border text-xs font-medium transition-colors ${weekdays.includes(String(i + 1)) ? "bg-primary text-primary-foreground border-primary" : "hover:bg-secondary"}`}>
   496→                  {d}
   497→                </button>
   498→              ))}
   499→            </div>
   500→          </div>
   501→        </div>
   502→        <DialogFooter>
   503→          <Button variant="outline" onClick={() => onOpenChange(false)}>Cancel</Button>
   504→          <Button onClick={submit} disabled={busy}>{busy ? "Saving…" : "Save service"}</Button>
   505→        </DialogFooter>
   506→      </DialogContent>
   507→    </Dialog>
   508→  );

   511→function EnrollDialog({ service, students, onOpenChange, onDone }: { service: Service | null; students: PosStudent[]; onOpenChange: (o: boolean) => void; onDone: () => void }) {
   512→  const { currentSchoolId } = useStore();
   513→  const [picked, setPicked] = useState<string[]>([]);
   514→  const [subsidy, setSubsidy] = useState("");
   515→  const [search, setSearch] = useState("");
   516→  const [busy, setBusy] = useState(false);
   517→
   518→  useEffect(() => { setPicked([]); setSearch(""); setSubsidy(""); }, [service]);
   519→
   520→  const list = useMemo(() => {
   521→    const q = search.toLowerCase();
   522→    return students.filter((s) => !q || s.name.toLowerCase().includes(q) || s.classroom.toLowerCase().includes(q));
   523→  }, [students, search]);
   524→
   525→  async function submit() {
   526→    if (!service || !currentSchoolId || !picked.length) return;
   527→    setBusy(true);
   528→    try {
   529→      const res = await apiPost<any>(`/api/school/${currentSchoolId}/ops/daily`, {
   530→        action: "enroll", serviceId: service.id, studentIds: picked,
   531→        priceCentsOverride: subsidy ? Math.round(parseFloat(subsidy) * 100) : undefined,
   532→      });
   533→      toast({ title: `${res.enrolled} learners enrolled`, description: `${service.name}${subsidy ? ` at GH₵${subsidy} subsidised rate` : ""}.` });
   534→      onDone();
   535→    } catch (e: any) { toast({ title: "Failed", description: e.message, variant: "destructive" }); } finally { setBusy(false); }
   536→  }
   537→
   538→  return (
   539→    <Dialog open={!!service} onOpenChange={onOpenChange}>
   540→      <DialogContent className="max-w-lg">
   541→        <DialogHeader>
   542→          <DialogTitle>Enrol learners — {service?.name}</DialogTitle>
   543→          <DialogDescription>Enrolled learners are billed automatically when you generate the day&apos;s charges.</DialogDescription>
   544→        </DialogHeader>
   545→        <div className="space-y-3">
   546→          <div className="relative">
   547→            <Search className="size-4 absolute left-2.5 top-2.5 text-muted-foreground" />
   548→            <Input placeholder="Search learner…" value={search} onChange={(e) => setSearch(e.target.value)} className="pl-8" />
   549→          </div>
   550→          <div className="border rounded-lg max-h-64 overflow-y-auto nice-scroll">
   551→            {list.map((s) => (
   552→              <label key={s.id} className="flex items-center gap-2.5 px-3 py-2 border-b last:border-0 hover:bg-secondary/40 cursor-pointer text-sm">
   553→                <input type="checkbox" checked={picked.includes(s.id)} onChange={() => setPicked((p) => (p.includes(s.id) ? p.filter((x) => x !== s.id) : [...p, s.id]))} className="accent-primary" />
   554→                <span className="flex-1">{s.name}</span>
   555→                <span className="text-xs text-muted-foreground">{s.classroom}</span>
   556→              </label>
   557→            ))}
   558→          </div>
   559→          <div className="space-y-1.5">
   560→            <Label>Subsidised rate (GH₵) — optional</Label>
   561→            <Input type="number" step="0.5" value={subsidy} onChange={(e) => setSubsidy(e.target.value)} placeholder={`Standard ${service ? fmtCedi(service.price) : ""}`} />
   562→            <p className="text-xs text-muted-foreground">For needy-family support or staff wards — overrides the standard price for these learners.</p>
   563→          </div>
   564→        </div>
   565→        <DialogFooter>
   566→          <Button variant="outline" onClick={() => onOpenChange(false)}>Cancel</Button>
   567→          <Button onClick={submit} disabled={busy || !picked.length}>{busy ? "Enrolling…" : `Enrol ${picked.length || ""} learner${picked.length === 1 ? "" : "s"}`}</Button>
   568→        </DialogFooter>
   569→      </DialogContent>
   570→    </Dialog>
   571→  );
   572→}
