   231→async function siblingNodeIds(nodeId: string): Promise<string[]> {
   232→  const all = await db.associationNode.findMany();
   233→  const out: string[] = [];
   234→  let cur = all.find((n) => n.id === nodeId);
   235→  while (cur?.parentId) {
   236→    out.push(cur.parentId);
   237→    cur = all.find((n) => n.id === cur!.parentId);
   238→  }
   239→  return out;
   240→}

   242→// ── POST /api/school/[id]/[resource] ──────────────────────────
   243→export async function POST(req: NextRequest, { params }: { params: Promise<{ id: string; resource: string }> }) {
   244→  const { id: schoolId, resource } = await params;
   245→  const user = await getUser(req);
   246→  if (!user) return NextResponse.json({ error: "Not authenticated" }, { status: 401 });
   247→  if (!canOperateSchool(user, schoolId)) return NextResponse.json({ error: "Forbidden — you cannot operate this school" }, { status: 403 });
   248→  const school = await db.school.findUnique({ where: { id: schoolId } });
   249→  if (!school) return NextResponse.json({ error: "School not found" }, { status: 404 });
   250→  const body = await req.json().catch(() => ({}));
   251→  const action: string = body.action;
   252→
   253→  switch (resource) {
   254→    // ── FEES ──────────────────────────────────────────────
   255→    case "fees": {
   256→      if (action === "record-payment") {
   257→        const { invoiceId, amountCents, method, reference, offline, clientId } = body;
   258→        if (clientId) {
   259→          const existing = await db.payment.findFirst({ where: { reference: `SYNC:${clientId}` } });
   260→          if (existing) return NextResponse.json({ payment: existing, duplicate: true });
   261→        }
   262→        const inv = await db.invoice.findUnique({ where: { id: invoiceId }, include: { student: true } });
   263→        if (!inv) return NextResponse.json({ error: "Invoice not found" }, { status: 404 });
   264→        const amt = Math.max(0, Math.min(Number(amountCents), inv.totalCents - inv.discountCents - inv.paidCents));
   265→        if (amt <= 0) return NextResponse.json({ error: "Amount exceeds balance or is invalid" }, { status: 400 });
   266→        const seq = (await db.payment.count({ where: { schoolId } })) + 1001;
   267→        const payment = await db.payment.create({
   268→          data: {
   269→            receiptNo: `RCP-2026-${String(seq).padStart(5, "0")}`, invoiceId, schoolId, amountCents: amt,
   270→            method: method ?? "CASH", reference: clientId ? `SYNC:${clientId}` : reference ?? null,
   271→            receivedByName: user.fullName, paidAt: new Date(body.paidAt ?? Date.now()),
   272→            offlineQueued: !!offline, syncedAt: offline ? new Date() : null,
   273→          },
   274→        });
   275→        const paid = inv.paidCents + amt;
   276→        await db.invoice.update({ where: { id: invoiceId }, data: { paidCents: paid, status: paid >= inv.totalCents - inv.discountCents ? "PAID" : "PARTIAL" } });
   277→        return NextResponse.json({ payment: { ...payment, student: `${inv.student.firstName} ${inv.student.lastName}` } });
   278→      }
   279→      if (action === "generate-invoices") {
   280→        const { classroomId } = body;
   281→        const term = await db.term.findFirst({ where: { isCurrent: true, academicYear: { schoolId } } });
   282→        if (!term) return NextResponse.json({ error: "No current term" }, { status: 400 });
   283→        const fs = await db.feeStructure.findFirst({ where: { classroomId, termId: term.id }, include: { items: true } });
   284→        if (!fs) return NextResponse.json({ error: "No fee structure for this class — set one up first" }, { status: 400 });
   285→        const enrolled = await db.enrollment.findMany({ where: { classroomId, status: "ACTIVE" } });
   286→        const existing = await db.invoice.findMany({ where: { termId: term.id, schoolId }, select: { studentId: true } });
   287→        const have = new Set(existing.map((e) => e.studentId));
   288→        let created = 0;
   289→        const seq = await db.invoice.count({ where: { schoolId } });
   290→        for (const e of enrolled) {
   291→          if (have.has(e.studentId)) continue;
   292→          const inv = await db.invoice.create({ data: { invoiceNo: `INV-2026-${String(seq + created + 1).padStart(4, "0")}`, studentId: e.studentId, schoolId, termId: term.id, classroomName: fs.classroomId ? (await db.classroom.findUnique({ where: { id: fs.classroomId } }))?.name ?? "" : "", totalCents: fs.totalCents, status: "UNPAID", dueDate: new Date(Date.now() + 14 * 864e5) } });
   293→          for (const it of fs.items.filter((i) => !i.isOptional)) await db.invoiceItem.create({ data: { invoiceId: inv.id, name: it.name, amountCents: it.amountCents } });
   294→          created++;
   295→        }
   296→        return NextResponse.json({ created });
   297→      }
   298→      if (action === "flag-defaulter") {
   299→        const { studentId, amountCents, description, type = "FEE_ARREARS", severity = "HIGH" } = body;
   300→        const flag = await db.studentFlag.create({ data: { studentId, schoolId, raisedByName: user.fullName, type, severity, status: "ACTIVE", amountOwedCents: Number(amountCents) || null, description: description ?? "Outstanding fees after repeated reminders." } });
   301→        return NextResponse.json({ flag });
   302→      }
   303→      if (action === "resolve-flag") {
   304→        const { flagId, note } = body;
   305→        const f = await db.studentFlag.update({ where: { id: flagId }, data: { status: "RESOLVED", resolvedAt: new Date(), resolutionNote: note ?? "Resolved.", resolvedByName: user.fullName } });
   306→        return NextResponse.json({ flag: f });
   307→      }
   308→    }
   309→
   310→    // ── ATTENDANCE ────────────────────────────────────────
   311→    case "attendance": {
   312→      if (action === "save") {
   313→        const { classroomId, date, records, offline, clientId } = body;
   314→        if (clientId) {
   315→          const existing = await db.attendanceSession.findFirst({ where: { classroomId, date: new Date(date), takenByName: { contains: `SYNC:${clientId}` } } });
   316→          if (existing) return NextResponse.json({ session: existing, duplicate: true });
   317→        }
   318→        const term = await db.term.findFirst({ where: { isCurrent: true, academicYear: { schoolId } } });
   319→        let session = await db.attendanceSession.findFirst({ where: { classroomId, date: new Date(date) } });
   320→        if (session) await db.attendanceRecord.deleteMany({ where: { sessionId: session.id } });
   321→        else session = await db.attendanceSession.create({ data: { classroomId, schoolId, termId: term!.id, date: new Date(date), takenByName: `${user.fullName}${clientId ? ` (SYNC:${clientId})` : ""}` } });
   322→        for (const r of records) {
   323→          await db.attendanceRecord.create({ data: { sessionId: session.id, studentId: r.studentId, status: r.status, markedOffline: !!offline } });
   324→        }
   325→        return NextResponse.json({ session: { id: session.id }, saved: records.length });
   326→      }
   327→    }
   328→
   329→    // ── GRADEBOOK ─────────────────────────────────────────
   330→    case "gradebook": {
   331→      if (action === "create-assessment") {
   332→        const { classroomId, subjectId, title, type, maxScore, date } = body;
   333→        const term = await db.term.findFirst({ where: { isCurrent: true, academicYear: { schoolId } } });
   334→        const a = await db.assessment.create({ data: { classroomId, schoolId, termId: term!.id, subjectId, title, type: type ?? "EXERCISE", maxScore: Number(maxScore) || 100, date: date ? new Date(date) : new Date() } });
   335→        return NextResponse.json({ assessment: a });
   336→      }
   337→      if (action === "save-marks") {
   338→        const { assessmentId, marks, offline } = body;
   339→        const a = await db.assessment.findUnique({ where: { id: assessmentId } });
   340→        if (!a) return NextResponse.json({ error: "Assessment not found" }, { status: 404 });
   341→        await db.mark.deleteMany({ where: { assessmentId } });
   342→        for (const m of marks) {
   343→          if (m.score !== null && m.score !== undefined && m.score !== "") {
   344→            await db.mark.create({ data: { assessmentId, studentId: m.studentId, score: Number(m.score), enteredOffline: !!offline } });
   345→          }
   346→        }
   347→        return NextResponse.json({ saved: marks.length });
   348→      }
   349→      if (action === "delete-assessment") {
   350→        await db.mark.deleteMany({ where: { assessmentId: body.assessmentId } });
   351→        await db.assessment.delete({ where: { id: body.assessmentId } });
   352→        return NextResponse.json({ ok: true });
   353→      }
   354→    }
   355→
   356→    // ── REPORT CARDS ──────────────────────────────────────
   357→    case "reports": {
   358→      if (action === "generate") {
   359→        const { classroomId, termId } = body;
   360→        const term = await db.term.findUnique({ where: { id: termId }, include: { academicYear: true } });
   361→        if (!term) return NextResponse.json({ error: "Term not found" }, { status: 404 });
   362→        const classroom = await db.classroom.findUnique({ where: { id: classroomId } });
   363→        const enrolled = await db.enrollment.findMany({ where: { classroomId, status: "ACTIVE" }, include: { student: true } });
   364→        const assessments = await db.assessment.findMany({ where: { classroomId, termId }, include: { subject: true, marks: true } });
   365→        if (!assessments.length) return NextResponse.json({ error: "No assessments recorded for this class and term yet — enter marks in the Gradebook first." }, { status: 400 });
   366→        // group by subject
   367→        const bySubject: Record<string, { subjectId: string; name: string; marks: Record<string, number>; max: number }> = {};
   368→        for (const a of assessments) {
   369→          const key = a.subject.name;
   370→          bySubject[key] ??= { subjectId: a.subjectId, name: key, marks: {}, max: 0 };
   371→          for (const m of a.marks) {
   372→            if (m.score === null) continue;
   373→            bySubject[key].marks[m.studentId] = (bySubject[key].marks[m.studentId] ?? 0) + m.score;
   374→          }
   375→          bySubject[key].max += a.maxScore;
   376→        }
   377→        const results: { studentId: string; subjects: { name: string; pct: number }[] }[] = enrolled.map((e) => ({
   378→          studentId: e.student.id,
   379→          subjects: Object.values(bySubject).map((s) => ({ name: s.name, pct: s.marks[e.student.id] !== undefined ? Math.round((s.marks[e.student.id] / s.max) * 100) : 0 })),
   380→        }));
   381→        const withAvg = results.map((r) => ({ ...r, avg: r.subjects.length ? r.subjects.reduce((s, x) => s + x.pct, 0) / r.subjects.length : 0 })).sort((a, b) => b.avg - a.avg);
   382→        let created = 0;
   383→        for (let i = 0; i < withAvg.length; i++) {
   384→          const r = withAvg[i];
   385→          const existing = await db.reportCard.findFirst({ where: { studentId: r.studentId, termId } });
   386→          if (existing) { await db.reportCardResult.deleteMany({ where: { reportCardId: existing.id } }); await db.reportCard.delete({ where: { id: existing.id } }); }
   387→          const rc = await db.reportCard.create({ data: { studentId: r.studentId, termId, classroomName: classroom?.name ?? "", totalScore: Math.round(r.avg * r.subjects.length), average: Math.round(r.avg), classPosition: i + 1, classSize: withAvg.length, promotion: r.avg >= 40 ? "PROMOTED" : "PROBATION", published: false } });
   388→          for (const s of r.subjects) {
   389→            const g = gradeFor(school.curriculum, s.pct);
   390→            await db.reportCardResult.create({ data: { reportCardId: rc.id, subjectName: s.name, total: s.pct, grade: `${g.grade} — ${g.label}`, remark: s.pct >= 68 ? "Commendable" : s.pct >= 55 ? "Good effort" : s.pct >= 40 ? "Fair — needs practice" : "Needs close support" } });
   391→          }
   392→          created++;
   393→        }
   394→        return NextResponse.json({ created });
   395→      }
   396→      if (action === "publish") {
   397→        await db.reportCard.update({ where: { id: body.reportCardId }, data: { published: true, publishedAt: new Date() } });
   398→        return NextResponse.json({ ok: true });
   399→      }
   400→      if (action === "publish-all") {
   401→        await db.reportCard.updateMany({ where: { termId: body.termId }, data: { published: true, publishedAt: new Date() } });
   402→        return NextResponse.json({ ok: true });
   403→      }
   404→      if (action === "update-remarks") {
   405→        const { reportCardId, teacherRemark, headRemark, conduct, promotion } = body;
   406→        const rc = await db.reportCard.update({ where: { id: reportCardId }, data: { teacherRemark, headRemark, conduct, promotion } });
   407→        return NextResponse.json({ reportCard: rc });
   408→      }
   409→    }
   410→
   411→    // ── STUDENTS ──────────────────────────────────────────
   412→    case "students": {
   413→      if (action === "create") {
   414→        const { firstName, lastName, otherNames, gender, dateOfBirth, guardianName, guardianPhone, guardianOccupation, relationship, classroomId, previousSchool, inquiryId } = body;
   415→        if (!firstName || !lastName || !guardianName || !guardianPhone || !classroomId) return NextResponse.json({ error: "Missing required fields" }, { status: 400 });
   416→        const classroom = await db.classroom.findUnique({ where: { id: classroomId } });
   417→        if (!classroom) return NextResponse.json({ error: "Classroom not found" }, { status: 404 });
   418→        const count = await db.student.count();
   419→        const birthYear = dateOfBirth ? new Date(dateOfBirth).getFullYear() : 2016;
   420→        const student = await db.student.create({ data: { nyansapoIndex: `NY-${birthYear}-${String(100000 + count + 1).slice(1)}`, firstName, lastName, otherNames, gender, dateOfBirth: dateOfBirth ? new Date(dateOfBirth) : null, guardianName, guardianPhone, guardianOccupation, relationship: relationship ?? "Parent", previousSchool, status: "ACTIVE" } });
   421→        const roll = (await db.enrollment.count({ where: { classroomId } })) + 1;
   422→        await db.enrollment.create({ data: { studentId: student.id, classroomId, schoolId, rollNo: roll, admissionDate: new Date(), status: "ACTIVE" } });
   423→        // auto-bill for current term
   424→        const term = await db.term.findFirst({ where: { isCurrent: true, academicYear: { schoolId } } });
   425→        if (term) {
   426→          const fs = await db.feeStructure.findFirst({ where: { classroomId, termId: term.id }, include: { items: true } });
   427→          if (fs) {
   428→            const seq = (await db.invoice.count({ where: { schoolId } })) + 1;
   429→            const inv = await db.invoice.create({ data: { invoiceNo: `INV-2026-${String(seq).padStart(4, "0")}`, studentId: student.id, schoolId, termId: term.id, classroomName: classroom.name, totalCents: fs.totalCents, status: "UNPAID", dueDate: new Date(Date.now() + 14 * 864e5) } });
   430→            for (const it of fs.items.filter((i) => !i.isOptional)) await db.invoiceItem.create({ data: { invoiceId: inv.id, name: it.name, amountCents: it.amountCents } });
   431→          }
   432→        }
   433→        if (inquiryId) await db.admissionInquiry.update({ where: { id: inquiryId }, data: { status: "ENROLLED", handledByName: user.fullName, matchedStudentId: student.id } });
   434→        const n = await db.enrollment.count({ where: { schoolId, status: "ACTIVE" } });
   435→        await db.school.update({ where: { id: schoolId }, data: { studentsOnRoll: n } });
   436→        return NextResponse.json({ student, enrolled: true });
   437→      }
   438→    }
   439→
   440→    // ── SETTINGS ──────────────────────────────────────────
   441→    case "settings": {
   442→      if (action === "update-school") {
   443→        const { curriculum, motto, phone, email, nasiaStatus, gnacopsStatus } = body;
   444→        const s = await db.school.update({ where: { id: schoolId }, data: { curriculum: curriculum ?? school.curriculum, motto: motto ?? school.motto, phone: phone ?? school.phone, email: email ?? school.email, nasiaStatus: nasiaStatus ?? school.nasiaStatus, gnacopsStatus: gnacopsStatus ?? school.gnacopsStatus } });
   445→        return NextResponse.json({ school: s });
   446→      }
   447→      if (action === "create-classroom") {
   448→        const { name, grade, stage, capacity } = body;
   449→        const c = await db.classroom.create({ data: { schoolId, name, gradeLevel: grade, stageLabel: stage, capacity: Number(capacity) || 30 } });
   450→        return NextResponse.json({ classroom: c });
   451→      }
   452→      if (action === "create-subject") {
   453→        const { name, code } = body;
   454→        const s = await db.subject.create({ data: { schoolId, name, code: (code ?? name.slice(0, 3)).toUpperCase(), curriculum: school.curriculum } });
   455→        return NextResponse.json({ subject: s });
   456→      }
   457→    }
   458→
   459→    // ── ANNOUNCEMENTS ─────────────────────────────────────
   460→    case "announcements": {
   461→      if (action === "create") {
   462→        const { title, body: bodyText, audience, category, priority } = body;
   463→        const a = await db.announcement.create({ data: { schoolId, title, body: bodyText, audience: audience ?? "PARENTS", category: category ?? "GENERAL", priority: priority ?? "NORMAL", postedByName: user.fullName } });
   464→        return NextResponse.json({ announcement: a });
   465→      }
   466→    }
   467→  }
   468→  return NextResponse.json({ error: `Unknown action ${resource}:${action}` }, { status: 400 });
   469→}
